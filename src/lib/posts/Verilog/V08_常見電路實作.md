# 第 8 章　常見電路實作

本章收錄 RTL 設計中最常用的電路單元，建議每個都配上 testbench 自己驗證。

## 8.1 除頻器

### 偶數除頻（輸出時脈）

```verilog
module clk_div #(parameter N = 4)(          // N 為偶數，輸出頻率 = clk / N
  input  wire clk, rst_n,
  output reg  clk_out
);
  reg [15:0] cnt;
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      cnt <= 16'd0;
      clk_out <= 1'b0;
    end else if (cnt == (N/2 - 1)) begin
      cnt <= 16'd0;
      clk_out <= ~clk_out;
    end else begin
      cnt <= cnt + 1'b1;
    end
  end
endmodule
```

### 使能脈衝（推薦做法）

在 FPGA/ASIC 中不建議產生新時脈，而是產生**每 N 個週期一次的使能脈衝**：

```verilog
module tick_gen #(parameter N = 1000)(
  input  wire clk, rst_n,
  output reg  tick
);
  reg [15:0] cnt;
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin cnt <= 16'd0; tick <= 1'b0; end
    else if (cnt == N - 1) begin cnt <= 16'd0; tick <= 1'b1; end
    else begin cnt <= cnt + 1'b1; tick <= 1'b0; end
  end
endmodule
```

其他暫存器仍使用原本的 `clk`，只在 `tick` 為 1 時更新：`if (tick) ...`

## 8.2 兩級正反器同步器

用於把**非同步單位元訊號**帶入某個時脈域（如外部按鈕）：

```verilog
module sync2 (
  input  wire clk, rst_n,
  input  wire async_in,
  output wire sync_out
);
  reg m, s;
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin m <= 1'b0; s <= 1'b0; end
    else        begin m <= async_in; s <= m; end
  end
  assign sync_out = s;
endmodule
```

⚠️ 只適用單一位元。多位元資料跨時脈域不可逐位元各自同步（見第 9 章）。

## 8.3 按鈕去彈跳

```verilog
module debounce #(parameter CNT_MAX = 20'd1_000_000)(
  input  wire clk, rst_n, noisy,
  output reg  clean
);
  reg        s0, s1;
  reg [19:0] cnt;

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      s0 <= 1'b0; s1 <= 1'b0; cnt <= 20'd0; clean <= 1'b0;
    end else begin
      s0 <= noisy;
      s1 <= s0;                               // 兩級同步
      if (s1 == clean) cnt <= 20'd0;          // 與目前輸出相同：重新計時
      else begin
        cnt <= cnt + 1'b1;                    // 不同：累計穩定時間
        if (cnt == CNT_MAX - 1) begin
          clean <= s1;
          cnt   <= 20'd0;
        end
      end
    end
  end
endmodule
```

輸入需**連續穩定** `CNT_MAX` 個週期，輸出才會改變。

## 8.4 PWM

```verilog
module pwm #(parameter W = 8)(
  input  wire         clk, rst_n,
  input  wire [W-1:0] duty,
  output reg          pwm_out
);
  reg [W-1:0] cnt;
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin cnt <= {W{1'b0}}; pwm_out <= 1'b0; end
    else begin
      cnt     <= cnt + 1'b1;
      pwm_out <= (cnt < duty);
    end
  end
endmodule
```

## 8.5 LFSR（偽隨機數產生器）

8 位元，多項式 x⁸ + x⁶ + x⁵ + x⁴ + 1：

```verilog
module lfsr8 (
  input  wire       clk, rst_n,
  output reg  [7:0] q
);
  wire fb = q[7] ^ q[5] ^ q[4] ^ q[3];
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) q <= 8'h01;                  // 種子不可為 0
    else        q <= {q[6:0], fb};
  end
endmodule
```

週期為 255（全 0 為鎖死狀態，必須避免）。

## 8.6 格雷碼轉換

```verilog
module bin2gray #(parameter W = 4)(
  input  wire [W-1:0] bin,
  output wire [W-1:0] gray
);
  assign gray = bin ^ (bin >> 1);
endmodule

module gray2bin #(parameter W = 4)(
  input  wire [W-1:0] gray,
  output reg  [W-1:0] bin
);
  integer i;
  always @(*) begin
    bin[W-1] = gray[W-1];
    for (i = W-2; i >= 0; i = i - 1)
      bin[i] = bin[i+1] ^ gray[i];
  end
endmodule
```

格雷碼相鄰值只差一位元，常用於**非同步 FIFO 指標**的跨時脈傳遞。

## 8.7 記憶體

### 單埠 RAM（同步讀）

```verilog
module sp_ram #(parameter AW = 4, DW = 8)(
  input  wire          clk, we,
  input  wire [AW-1:0] addr,
  input  wire [DW-1:0] din,
  output reg  [DW-1:0] dout
);
  reg [DW-1:0] mem [0:(1<<AW)-1];

  always @(posedge clk) begin
    if (we) mem[addr] <= din;
    dout <= mem[addr];                 // 讀到的是寫入前的舊值 (read-first)
  end
endmodule
```

此寫法在多數 FPGA 工具中可被推導為 Block RAM。

### ROM（以 `$readmemh` 初始化）

```verilog
module rom #(parameter AW = 8, DW = 8)(
  input  wire          clk,
  input  wire [AW-1:0] addr,
  output reg  [DW-1:0] data
);
  reg [DW-1:0] mem [0:(1<<AW)-1];
  initial $readmemh("rom.hex", mem);
  always @(posedge clk) data <= mem[addr];
endmodule
```

## 8.8 同步 FIFO

```verilog
module sync_fifo #(
  parameter DEPTH = 16,
  parameter WIDTH = 8
)(
  input  wire             clk, rst_n,
  input  wire             wr_en, rd_en,
  input  wire [WIDTH-1:0] wdata,
  output reg  [WIDTH-1:0] rdata,
  output wire             full, empty
);
  localparam AW = $clog2(DEPTH);

  reg [WIDTH-1:0] mem [0:DEPTH-1];
  reg [AW:0]      wptr, rptr;           // 多 1 位元用來區分「滿」與「空」

  assign empty = (wptr == rptr);
  assign full  = (wptr[AW-1:0] == rptr[AW-1:0]) && (wptr[AW] != rptr[AW]);

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      wptr  <= {(AW+1){1'b0}};
      rptr  <= {(AW+1){1'b0}};
      rdata <= {WIDTH{1'b0}};
    end else begin
      if (wr_en && !full) begin
        mem[wptr[AW-1:0]] <= wdata;
        wptr <= wptr + 1'b1;
      end
      if (rd_en && !empty) begin
        rdata <= mem[rptr[AW-1:0]];
        rptr  <= rptr + 1'b1;
      end
    end
  end
endmodule
```

### 對應的簡單 Testbench

```verilog
`timescale 1ns/1ps
module tb_fifo;
  reg clk = 0, rst_n = 0, wr_en = 0, rd_en = 0;
  reg  [7:0] wdata = 0;
  wire [7:0] rdata;
  wire full, empty;

  sync_fifo #(.DEPTH(8), .WIDTH(8)) dut (
    .clk(clk), .rst_n(rst_n), .wr_en(wr_en), .rd_en(rd_en),
    .wdata(wdata), .rdata(rdata), .full(full), .empty(empty));

  always #5 clk = ~clk;

  reg [7:0] model [0:255];
  integer wp = 0, rp = 0, errors = 0, i;

  initial begin
    $dumpfile("wave.vcd"); $dumpvars(0, tb_fifo);
    #22 rst_n = 1;

    // 寫滿
    for (i = 0; i < 10; i = i + 1) begin
      @(negedge clk);
      if (!full) begin wr_en = 1; wdata = i + 8'h10; model[wp] = wdata; wp = wp + 1; end
      else wr_en = 0;
    end
    @(negedge clk) wr_en = 0;

    // 讀空
    for (i = 0; i < 10; i = i + 1) begin
      @(negedge clk);
      if (!empty) begin
        rd_en = 1;
        @(posedge clk); #1;                    // 讀出資料在此邊緣後有效
        if (rdata !== model[rp]) begin
          errors = errors + 1;
          $display("ERR: got %h exp %h", rdata, model[rp]);
        end
        rp = rp + 1;
      end else rd_en = 0;
    end
    @(negedge clk) rd_en = 0;

    if (errors == 0) $display("FIFO PASSED"); else $display("FIFO FAILED");
    $finish;
  end
endmodule
```

## 8.9 設計小技巧

- 計數器、狀態機的寬度用 `$clog2` 計算，避免手算錯誤
- 所有「每 N 週期做一次」的事用**使能脈衝**實作
- 參數化寬度與深度，方便重用
- 每個模組一個明確的功能，用 `u_xxx` 實例名稱
